What does this print? type Shape = { name: string; children: Shape[] }; const city: Shape = { name: "City", children: [] }; const state: Shape = { name: "State", children: Array(6).fill(city) }; const region: Shape = { name: "Region", children: Array(6).fill(state) }; const settings: Shape = { name: "Settings", children: Array(6).fill(region) }; let calls = 0; function size(shape: Shape): number { calls++; return 1 + shape.children.reduce((sum, child) => sum + size(child), 0); } const cache = new Map(); function sizeCached(shape: Shape): number { const known = cache.get(shape); if (known !== undefined) return known; calls++; const result = 1 + shape.children.reduce((sum, child) => sum + sizeCached(child), 0); cache.set(shape, result); return result; } console.log(size(settings), calls); calls = 0; console.log(sizeCached(settings), calls);

 A runtime analogy for tail position: recursion that nests versus work done in a loop. What does this print? function dashesNested(n: number): string { return n === 0 ?
